62enum PCD_Register {
63 // Page 0: Command and status
64 // 0x00 // reserved for future use
65 CommandReg = 0x01 << 1, // starts and stops command execution
66 ComIEnReg = 0x02 << 1, // enable and disable interrupt request control bits
67 DivIEnReg = 0x03 << 1, // enable and disable interrupt request control bits
68 ComIrqReg = 0x04 << 1, // interrupt request bits
69 DivIrqReg = 0x05 << 1, // interrupt request bits
70 ErrorReg = 0x06 << 1, // error bits showing the error status of the last
71 // command executed
72 Status1Reg = 0x07 << 1, // communication status bits
73 Status2Reg = 0x08 << 1, // receiver and transmitter status bits
74 FIFODataReg = 0x09 << 1, // input and output of 64 byte FIFO buffer
75 FIFOLevelReg = 0x0A << 1, // number of bytes stored in the FIFO buffer
76 WaterLevelReg = 0x0B << 1, // level for FIFO underflow and overflow warning
77 ControlReg = 0x0C << 1, // miscellaneous control registers
78 BitFramingReg = 0x0D << 1, // adjustments for bit-oriented frames
79 CollReg = 0x0E << 1, // bit position of the first bit-collision detected on
80 // the RF interface
81 // 0x0F // reserved for future use
82
83 // Page 1: Command
84 // 0x10 // reserved for future use
85 ModeReg = 0x11 << 1, // defines general modes for transmitting and receiving
86 TxModeReg = 0x12 << 1, // defines transmission data rate and framing
87 RxModeReg = 0x13 << 1, // defines reception data rate and framing
88 TxControlReg = 0x14 << 1, // controls the logical behavior of the antenna
89 // driver pins TX1 and TX2
90 TxASKReg = 0x15 << 1, // controls the setting of the transmission modulation
91 TxSelReg = 0x16 << 1, // selects the internal sources for the antenna driver
92 RxSelReg = 0x17 << 1, // selects internal receiver settings
93 RxThresholdReg = 0x18 << 1, // selects thresholds for the bit decoder
94 DemodReg = 0x19 << 1, // defines demodulator settings
95 // 0x1A // reserved for future use
96 // 0x1B // reserved for future use
97 MfTxReg =
98 0x1C << 1, // controls some MIFARE communication transmit parameters
99 MfRxReg =
100 0x1D << 1, // controls some MIFARE communication receive parameters
101 // 0x1E // reserved for future use
102 SerialSpeedReg =
103 0x1F << 1, // selects the speed of the serial UART interface
104
105 // Page 2: Configuration
106 // 0x20 // reserved for future use
107 CRCResultRegH =
108 0x21 << 1, // shows the MSB and LSB values of the CRC calculation
109 CRCResultRegL = 0x22 << 1,
110 // 0x23 // reserved for future use
111 ModWidthReg = 0x24 << 1, // controls the ModWidth setting?
112 // 0x25 // reserved for future use
113 RFCfgReg = 0x26 << 1, // configures the receiver gain
114 GsNReg = 0x27 << 1, // selects the conductance of the antenna driver pins
115 // TX1 and TX2 for modulation
116 CWGsPReg = 0x28 << 1, // defines the conductance of the p-driver output
117 // during periods of no modulation
118 ModGsPReg = 0x29 << 1, // defines the conductance of the p-driver output
119 // during periods of modulation
120 TModeReg = 0x2A << 1, // defines settings for the internal timer
121 TPrescalerReg = 0x2B << 1, // the lower 8 bits of the TPrescaler value. The
122 // 4 high bits are in TModeReg.
123 TReloadRegH = 0x2C << 1, // defines the 16-bit timer reload value
124 TReloadRegL = 0x2D << 1,
125 TCounterValueRegH = 0x2E << 1, // shows the 16-bit timer value
126 TCounterValueRegL = 0x2F << 1,
127
128 // Page 3: Test Registers
129 // 0x30 // reserved for future use
130 TestSel1Reg = 0x31 << 1, // general test signal configuration
131 TestSel2Reg = 0x32 << 1, // general test signal configuration
132 TestPinEnReg = 0x33 << 1, // enables pin output driver on pins D1 to D7
133 TestPinValueReg =
134 0x34
135 << 1, // defines the values for D1 to D7 when it is used as an I/O bus
136 TestBusReg = 0x35 << 1, // shows the status of the internal test bus
137 AutoTestReg = 0x36 << 1, // controls the digital self-test
138 VersionReg = 0x37 << 1, // shows the software version
139 AnalogTestReg = 0x38 << 1, // controls the pins AUX1 and AUX2
140 TestDAC1Reg = 0x39 << 1, // defines the test value for TestDAC1
141 TestDAC2Reg = 0x3A << 1, // defines the test value for TestDAC2
142 TestADCReg = 0x3B << 1 // shows the value of ADC I and Q channels
143 // 0x3C // reserved for production tests
144 // 0x3D // reserved for production tests
145 // 0x3E // reserved for production tests
146 // 0x3F // reserved for production tests
147};

152typedef enum _PCD_Command {

154 PCD_Mem = 0x01, // stores 25 bytes into the internal buffer
155 PCD_GenerateRandomID = 0x02, // generates a 10-byte random ID number

157 PCD_Transmit = 0x04, // transmits data from the FIFO buffer

159 // CommandReg register bits without affecting the

166 0x0E, // performs the MIFARE standard authentication as a reader

169} PCD_Command;

197typedef enum _PICC_Command {
198 // The commands used by the PCD to manage communication with several PICCs
199 // (ISO 14443-3, Type A, section 6.4)
200 PICC_CMD_REQA = 0x26, // REQuest command, Type A. Invites PICCs in state
201 // IDLE to go to READY and prepare for anti collision
202 // or selection. 7 bit frame.
203 PICC_CMD_WUPA = 0x52, // Wake-UP command, Type A. Invites PICCs in state
204 // IDLE and HALT to go to READY(*) and prepare for
205 // anti collision or selection. 7 bit frame.
206 PICC_CMD_CT = 0x88, // Cascade Tag. Not really a command, but used during
207 // anti collision.
208 PICC_CMD_SEL_CL1 = 0x93, // Anti collision/Select, Cascade Level 1
209 PICC_CMD_SEL_CL2 = 0x95, // Anti collision/Select, Cascade Level 2
210 PICC_CMD_SEL_CL3 = 0x97, // Anti collision/Select, Cascade Level 3
211 PICC_CMD_HLTA = 0x50, // HaLT command, Type A. Instructs an ACTIVE PICC to
212 // go to state HALT.
213 // The commands used for MIFARE Classic (from
214 // http://www.mouser.com/ds/2/302/MF1S503x-89574.pdf, Section 9)
215 // Use PCD_MFAuthent to authenticate access to a sector, then use these
216 // commands to read/write/modify the blocks on the sector.
217 // The read/write commands can also be used for MIFARE Ultralight.
218 PICC_CMD_MF_AUTH_KEY_A = 0x60, // Perform authentication with Key A
219 PICC_CMD_MF_AUTH_KEY_B = 0x61, // Perform authentication with Key B
220 PICC_CMD_MF_READ = 0x30, // Reads one 16 byte block from the
221 // authenticated sector of the PICC. Also used for
222 // MIFARE Ultralight.
223 PICC_CMD_MF_WRITE = 0xA0, // Writes one 16 byte block to the
224 // authenticated sector of the PICC. Called
225 // "COMPATIBILITY WRITE" for MIFARE Ultralight.
226 PICC_CMD_MF_DECREMENT = 0xC0, // Decrements the contents of a block and
227 // stores the result in the internal data
228 // register.
229 PICC_CMD_MF_INCREMENT = 0xC1, // Increments the contents of a block and
230 // stores the result in the internal data
231 // register.
232 PICC_CMD_MF_RESTORE =
233 0xC2, // Reads the contents of a block into the internal data register.
234 PICC_CMD_MF_TRANSFER =
235 0xB0, // Writes the contents of the internal data register to a block.
236 // The commands used for MIFARE Ultralight (from
237 // http://www.nxp.com/documents/data_sheet/MF0ICU1.pdf, Section 8.6)
238 // The PICC_CMD_MF_READ and PICC_CMD_MF_WRITE can also be used for MIFARE
239 // Ultralight.
240 PICC_CMD_UL_WRITE = 0xA2 // Writes one 4 byte page to the PICC.
241} PICC_Command;
